        function toggleChat() {
            const drawer = document.getElementById('chat-drawer');
            drawer.classList.toggle('translate-x-full');
        }

        // Toggle micro-interactions for Mic/Cam
        const micBtn = document.getElementById('mic-toggle');
        const camBtn = document.getElementById('cam-toggle');

        micBtn.addEventListener('click', () => {
            const icon = micBtn.querySelector('.material-symbols-outlined');
            if (icon.innerText === 'mic') {
                icon.innerText = 'mic_off';
                micBtn.classList.add('text-error');
                micBtn.classList.remove('text-primary-container');
            } else {
                icon.innerText = 'mic';
                micBtn.classList.remove('text-error');
                micBtn.classList.add('text-primary-container');
            }
        });

        camBtn.addEventListener('click', () => {
            const icon = camBtn.querySelector('.material-symbols-outlined');
            if (icon.innerText === 'videocam') {
                icon.innerText = 'videocam_off';
                camBtn.classList.add('text-error');
                camBtn.classList.remove('text-primary-container');
            } else {
                icon.innerText = 'videocam';
                camBtn.classList.remove('text-error');
                camBtn.classList.add('text-primary-container');
            }
        });

        // Hide UI on inactivity (Atmospheric Effect)
        let timeout;
        const hideUI = () => {
            document.querySelectorAll('header, .fixed.bottom-stack-lg, .fixed.bottom-32').forEach(el => {
                el.style.opacity = '0';
                el.style.pointerEvents = 'none';
                el.style.transition = 'opacity 0.8s ease';
            });
        };

        const showUI = () => {
            document.querySelectorAll('header, .fixed.bottom-stack-lg, .fixed.bottom-32').forEach(el => {
                el.style.opacity = '1';
                el.style.pointerEvents = 'auto';
            });
            clearTimeout(timeout);
            timeout = setTimeout(hideUI, 5000);
        };

        window.addEventListener('mousemove', showUI);
        window.addEventListener('touchstart', showUI);
        timeout = setTimeout(hideUI, 5000);
    </script>

        // Micro-interaction for typing indicator simulation
        const chatScroller = document.getElementById('chat-scroller');
        const typingIndicator = document.getElementById('typing-indicator');

        function simulateTyping() {
            setTimeout(() => {
                typingIndicator.style.opacity = '1';
                chatScroller.scrollTop = chatScroller.scrollHeight;
                
                setTimeout(() => {
                    typingIndicator.style.opacity = '0';
                }, 3000);
            }, 2000);
        }

        // Auto-scroll to bottom on load
        window.onload = () => {
            chatScroller.scrollTop = chatScroller.scrollHeight;
            simulateTyping();
        };

        // Subtle parallax effect on message bubbles on scroll
        chatScroller.addEventListener('scroll', () => {
            const bubbles = document.querySelectorAll('.matte-glow');
            const scrollPos = chatScroller.scrollTop;
            
            bubbles.forEach((bubble, index) => {
                const speed = 0.02;
                const yPos = (scrollPos * speed * (index % 2 === 0 ? 1 : -1));
                bubble.style.transform = `translateY(${yPos}px)`;
            });
        });
